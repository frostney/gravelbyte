cmake_minimum_required(VERSION 3.16)
option(GRAVELBYTE_DIAGNOSTICS "Enable developer FPS overlays and diagnostic controls" OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(GRAVELBYTE_DEVICE "Build PicoSystem firmware" OFF)
option(GRAVELBYTE_SMOKE "Exercise real player menu and flash writes on hardware" OFF)
option(GRAVELBYTE_BENCHMARK "Drive the stage automatically on hardware without saving a best" OFF)
if(GRAVELBYTE_DEVICE)
  if(GRAVELBYTE_SMOKE AND GRAVELBYTE_BENCHMARK)
    message(FATAL_ERROR "Choose one device diagnostic at a time")
  endif()
  set(PICOSYSTEM_DIR "$ENV{PICOSYSTEM_DIR}" CACHE PATH "Official PicoSystem SDK checkout")
  if(NOT EXISTS "${PICOSYSTEM_DIR}/libraries/picosystem.cmake")
    message(FATAL_ERROR "Set PICOSYSTEM_DIR to the official PicoSystem SDK checkout")
  endif()
  if(NOT DEFINED PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Raspberry Pi Pico SDK checkout")
  endif()
  set(PICO_BOARD pimoroni_picosystem CACHE STRING "PicoSystem board" FORCE)
  include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(gravelbyte LANGUAGES C CXX ASM)
option(GRAVELBYTE_TEST_SILENT "Silence platform audio during local device testing" OFF)
if(GRAVELBYTE_TEST_SILENT)
  add_compile_definitions(GRAVELBYTE_TEST_SILENT=1)
endif()
if(GRAVELBYTE_DIAGNOSTICS)
  add_compile_definitions(GRAVELBYTE_DIAGNOSTICS=1)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(GRAVELBYTE_DEVICE)
  pico_sdk_init()
  include(${PICOSYSTEM_DIR}/libraries/picosystem.cmake)
  picosystem_executable(gravelbyte src/pico_main.cpp src/game.cpp src/stage.cpp src/render.cpp)
  pixel_double(gravelbyte)
  no_spritesheet(gravelbyte)
  no_font(gravelbyte)
  disable_startup_logo(gravelbyte)
  target_link_libraries(gravelbyte hardware_flash hardware_sync pico_flash)
  target_compile_options(gravelbyte PRIVATE -O3 -Wall -Wextra)
  target_compile_definitions(gravelbyte PRIVATE GRAVELBYTE_PROFILE=1)
  if(GRAVELBYTE_SMOKE)
    target_compile_definitions(gravelbyte PRIVATE GRAVELBYTE_SMOKE=1)
  endif()
  if(GRAVELBYTE_BENCHMARK)
    set(GRAVELBYTE_BENCHMARK_START 0 CACHE STRING "First diagnostic car/track pair (0-8)")
    if(GRAVELBYTE_BENCHMARK_START LESS 0 OR GRAVELBYTE_BENCHMARK_START GREATER 8)
      message(FATAL_ERROR "GRAVELBYTE_BENCHMARK_START must be 0-8")
    endif()
    target_compile_definitions(gravelbyte PRIVATE GRAVELBYTE_BENCHMARK=1 GRAVELBYTE_BENCHMARK_START=${GRAVELBYTE_BENCHMARK_START})
  endif()
  pico_enable_stdio_usb(gravelbyte 1)
  pico_enable_stdio_uart(gravelbyte 0)
elseif(EMSCRIPTEN)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    OUTPUT_VARIABLE GRAVELBYTE_BUILD_ID OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GRAVELBYTE_GIT_RESULT)
  if(NOT GRAVELBYTE_GIT_RESULT EQUAL 0)
    message(FATAL_ERROR "Build the web release from a Git checkout so assets have a revision identity")
  endif()
  add_executable(gravelbyte src/web_main.cpp src/game.cpp src/stage.cpp src/render.cpp)
  target_compile_options(gravelbyte PRIVATE -O3 -Wall -Wextra)
  target_compile_definitions(gravelbyte PRIVATE GRAVELBYTE_BUILD_ID="${GRAVELBYTE_BUILD_ID}")
  target_link_options(gravelbyte PRIVATE -O3 --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createGravelbyte -sENVIRONMENT=web -sFILESYSTEM=0 -sALLOW_MEMORY_GROWTH=0 -sINITIAL_MEMORY=16777216 -sEXPORTED_RUNTIME_METHODS=HEAPU8,HEAPU16,UTF8ToString)
  set_target_properties(gravelbyte PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/web-raw")
  add_custom_target(package-web ALL
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/package_site.py
      --source ${CMAKE_SOURCE_DIR}/site --raw ${CMAKE_BINARY_DIR}/web-raw
      --output ${CMAKE_BINARY_DIR}/site --build-id ${GRAVELBYTE_BUILD_ID}
    DEPENDS gravelbyte VERBATIM)
else()
  add_library(rally src/game.cpp src/stage.cpp src/render.cpp)
  target_include_directories(rally PUBLIC src)
  target_compile_options(rally PRIVATE -Wall -Wextra -Wpedantic)
  find_package(SDL2 REQUIRED)
  add_executable(gravelbyte src/desktop_main.cpp)
  target_link_libraries(gravelbyte PRIVATE rally SDL2::SDL2)
  if(APPLE)
    set_target_properties(gravelbyte PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_GUI_IDENTIFIER "local.gravelbyte.preview"
      MACOSX_BUNDLE_BUNDLE_NAME "gravelbyte"
      MACOSX_BUNDLE_BUNDLE_VERSION "0.1.0"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1")
  endif()
  enable_testing()
  add_executable(authored_tests tests/authored_tests.cpp)
  target_link_libraries(authored_tests PRIVATE rally)
  add_test(NAME authored_courses_and_driving COMMAND authored_tests)
  add_executable(cinematic_tests tests/cinematic_tests.cpp)
  target_link_libraries(cinematic_tests PRIVATE rally)
  target_compile_definitions(cinematic_tests PRIVATE GRAVELBYTE_V1_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/records-v1.bin")
  add_test(NAME cinematics_progression_landmarks COMMAND cinematic_tests)
  add_executable(tree_tests tests/tree_tests.cpp)
  target_link_libraries(tree_tests PRIVATE rally)
  add_test(NAME snow_tree_visibility COMMAND tree_tests)
  add_executable(shadow_tests tests/shadow_tests.cpp)
  target_link_libraries(shadow_tests PRIVATE rally)
  add_test(NAME shadow_surface_transitions COMMAND shadow_tests)
  add_executable(features_tests tests/features_tests.cpp)
  target_link_libraries(features_tests PRIVATE rally)
  add_test(NAME cars_records_shadows COMMAND features_tests)
  add_executable(rally_tests tests/game_tests.cpp)
  target_link_libraries(rally_tests PRIVATE rally)
  add_test(NAME rally_contracts COMMAND rally_tests)
  add_executable(stage_drive tests/stage_drive.cpp)
  target_link_libraries(stage_drive PRIVATE rally)
  add_test(NAME complete_stage_drive COMMAND stage_drive)
  add_executable(failure_tests tests/failure_tests.cpp)
  target_link_libraries(failure_tests PRIVATE rally)
  add_test(NAME failure_paths COMMAND failure_tests)
  add_executable(terrain_tests tests/terrain_tests.cpp)
  target_link_libraries(terrain_tests PRIVATE rally)
  add_test(NAME terrain_clearance COMMAND terrain_tests)
endif()

include(cmake/Format.cmake)
